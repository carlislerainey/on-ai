% ---- Margin notes -----------------------------------------------------------
% \footnote{...} becomes a numbered note in the margin, beside the line that
% cites it; \aside{...} is an unnumbered one. Notes are single spaced, stack
% without overlapping, and move to the next page when a page's margin is
% full (marginfix). They cannot go inside floats, tables, or minipages.
% Title-page notes (\titlenote, \addauthor) stay at the foot of the page.

% ---- Appendix and supplement ------------------------------------------------
% \appendix: sections become A, B, ...; figures, tables, and equations
% become A1, A2, ... within each section.
\renewcommand{\appendix}{%
  \par
  \setcounter{section}{0}%
  \setcounter{secnumdepth}{3}%
  \renewcommand{\thesection}{\Alph{section}}%
  \counterwithin*{figure}{section}%
  \counterwithin*{table}{section}%
  \counterwithin*{equation}{section}%
  \renewcommand{\thefigure}{\thesection\arabic{figure}}%
  \renewcommand{\thetable}{\thesection\arabic{table}}%
  \renewcommand{\theequation}{\thesection\arabic{equation}}}

\newcommand{\email}[1]{\href{mailto:#1}{#1}}

\newcounter{rp@nauthors}
% \addauthor[line under the name]{Name}{footnote text}
\newcommand{\addauthor}[3][]{%
  \stepcounter{rp@nauthors}%
  \ifnum\value{rp@nauthors}=1
    \g@addto@macro\rp@pdfauthors{#2}%
  \else
    \g@addto@macro\rp@pdfauthors{, #2}%
  \fi
  \g@addto@macro\rp@authors{\rp@oneauthor{#1}{#2}{#3}}}
\newcommand{\rp@oneauthor}[3]{%
  \par\vspace{0.9ex}%
  {\sfweight{regular}\large #2}%
  \if\relax\detokenize{#3}\relax\else\rp@symbolnote{#3}\fi
  \if\relax\detokenize{#1}\relax\else\par{\sfweight{light}\small #1}\fi
  \par}

% Footnote marked *, dagger, ... that does not disturb the main counter.
\newcounter{rp@symbol}
\newcommand{\rp@symbolnote}[1]{%
  \stepcounter{rp@symbol}%
  \begingroup
    \renewcommand{\thefootnote}{\fnsymbol{footnote}}%
    \setcounter{footnote}{\value{rp@symbol}}%
    \addtocounter{footnote}{-1}%
    \pagefootnote{#1}%
  \endgroup}

\renewcommand{\maketitle}{%
  \hypersetup{pdftitle={\@title}}%
  \ifanonymous\else\hypersetup{pdfauthor={\rp@pdfauthors}}\fi
  \iftitlepage\thispagestyle{empty}\fi
  \begingroup
    \singlespacing
    \RaggedRight
    \setlength{\parskip}{0pt}%
    \iftitlepage\vspace*{0.6in}\else\vspace*{-0.2in}\fi
    \begin{fullwidth}\RaggedRight
    {\headingfont\huge\@title
      \ifanonymous\else
        \ifx\rp@titlenote\@empty\else\rp@symbolnote{\rp@titlenote}\fi
      \fi\par}%
    \ifx\rp@subtitle\@empty\else
      \vspace{1.2ex}{\sfweight{light}\Large\rp@subtitle\par}%
    \fi
    \end{fullwidth}
    \vspace{0.2in}%
    \ifanonymous\else
      \rp@authors
      \vspace{0.12in}%
    \fi
    {\sfweight{light}\@date\par}%
    \ifanonymous\else
      \ifx\rp@versionnote\@empty\else
        \vspace{1ex}{\small\rp@versionnote\par}%
      \fi
    \fi
    \vspace{0.2in}%
  \endgroup
  \setcounter{footnote}{0}%
  \iftitlepage\else\bodyspacing\fi}

\renewenvironment{abstract}{%
  \begingroup
  \iftitlepage\singlespacing\fi
  \noindent{\headingfont\normalsize Abstract\par}\vspace{0.8ex}%
  \list{}{\leftmargin=0pt \rightmargin=0pt}\item\relax
  \setstretch{1.15}\noindent\ignorespaces}{%
  \endlist
  \ifx\rp@keywords\@empty\else
    \par\vspace{1.5ex}{\RaggedRight\small
      {\headingfont Keywords:} \rp@keywords\par}%
  \fi
  \ifx\rp@wordcount\@empty\else
    \par\vspace{1ex}{\RaggedRight\small Word count: \rp@wordcount\par}%
  \fi
  \endgroup
  \iftitlepage
    \clearpage
    \setcounter{page}{1}%
    \bodyspacing
  \else
    \par\vspace{2ex}%
  \fi}
\makeatother
